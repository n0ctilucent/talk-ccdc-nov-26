\section{Format \& Rules}

\begin{frame}{Battle Structure}
  \begin{enumerate}
    \item \hl{Round 1} — Theme revealed; participants interpret freely
    \item \hl{Round 2} — Constraint round (style, medium, palette)
    \item \hl{Final Round} — Blind prompt; judge's pick + audience vote
  \end{enumerate}

  \vspace{-0.3cm}
  {\footnotesize Timed: ~15 min per round, ~30 s upload window}
\end{frame}

\begin{frame}{Rules of Engagement}
  \begin{itemize}
    \item AI-generated only — no hand editing after generation
    \item All tools/models allowed (open-source or proprietary)
    \item One submission per round (highest quality)
    \item Prompt must be shared publicly with each submission
  \end{itemize}

  \note{Enforce: no inpainting/outpainting that adds content. Style transfer is fine.}
\end{frame}
